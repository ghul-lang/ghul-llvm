cmake_minimum_required(VERSION 3.20)

project(ghul-llvm LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# LLVM_ROOT names the LLVM install to build against: build/tool.sh passes
# LLVM's release, unpacked down to its headers and static libraries. The
# release's own CMake package is not used, because it insists that every tool
# it was built with is present beside the libraries.
set(LLVM_ROOT "" CACHE PATH "the LLVM install to build against")

if(NOT EXISTS "${LLVM_ROOT}/include/llvm/Config/llvm-config.h")
    message(FATAL_ERROR "LLVM_ROOT does not name an LLVM install: ${LLVM_ROOT}")
endif()

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" version_file)
string(STRIP "${version_file}" version_file)

set(GHUL_LLVM_VERSION "${version_file}" CACHE STRING "the version ghul-llvm reports")
set(GHUL_LLVM_TRIPLE "x86_64-unknown-linux-musl" CACHE STRING "the triple ghul-llvm generates code for")
set(GHUL_LLVM_BACKEND "X86" CACHE STRING "the LLVM backend for that triple, as LLVM names it")

# LLD's driver initialises every backend LLVM was built with, for link-time
# optimisation, and LLVM's release is built with all of them. Each backend
# but ours gets initialisers that do nothing, defined here, so the linker
# never has a reason to take that backend out of its archive. This is most
# of the difference between a 30MB executable and a 120MB one.
file(STRINGS "${LLVM_ROOT}/include/llvm/Config/Targets.def" target_lines REGEX "^LLVM_TARGET\\(")
set(backend_stubs "// Generated by CMakeLists.txt.\n\nextern \"C\" {\n")

foreach(line IN LISTS target_lines)
    string(REGEX REPLACE "^LLVM_TARGET\\(([A-Za-z0-9_]+)\\).*" "\\1" backend "${line}")

    if(NOT backend STREQUAL GHUL_LLVM_BACKEND)
        foreach(part TargetInfo Target TargetMC AsmParser AsmPrinter Disassembler TargetMCA)
            string(APPEND backend_stubs "void LLVMInitialize${backend}${part}() {}\n")
        endforeach()
    endif()
endforeach()

string(APPEND backend_stubs "}\n")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/backend_stubs.cpp" CONTENT "${backend_stubs}" @ONLY)

add_executable(ghul-llvm
    src/compile.cpp
    src/link.cpp
    src/main.cpp
    src/options.cpp
    src/polly.cpp
    "${CMAKE_CURRENT_BINARY_DIR}/backend_stubs.cpp"
)

target_include_directories(ghul-llvm SYSTEM PRIVATE "${LLVM_ROOT}/include")

# The definitions and flags LLVM's headers are built with, as llvm-config
# --cxxflags reports them for the release.
target_compile_definitions(ghul-llvm PRIVATE
    _GNU_SOURCE
    __STDC_CONSTANT_MACROS
    __STDC_FORMAT_MACROS
    __STDC_LIMIT_MACROS
    GHUL_LLVM_VERSION="${GHUL_LLVM_VERSION}"
    GHUL_LLVM_TRIPLE="${GHUL_LLVM_TRIPLE}"
)

target_compile_options(ghul-llvm PRIVATE -fno-exceptions -fno-rtti -Wall -Wextra)

# Every LLVM and LLD archive, as one group: the linker takes from them only
# the members something refers to, so which libraries are needed, and in what
# order, never has to be written down.
file(GLOB llvm_archives "${LLVM_ROOT}/lib/libLLVM*.a" "${LLVM_ROOT}/lib/liblld*.a")

target_link_libraries(ghul-llvm PRIVATE
    -Wl,--start-group ${llvm_archives} -Wl,--end-group
)

# Everything but the C library is linked in: the C++ runtime, and zlib and zstd,
# which LLVM's release libraries use. Sections nothing reaches are dropped.
target_link_options(ghul-llvm PRIVATE
    -static-libstdc++
    -static-libgcc
    -Wl,--gc-sections
    -Wl,--as-needed
)

target_link_libraries(ghul-llvm PRIVATE -Wl,-Bstatic z zstd -Wl,-Bdynamic pthread dl m rt)

install(TARGETS ghul-llvm RUNTIME DESTINATION bin)
